// TerminalConfig configures a new terminal. Cols and Rows must be nonzero.
// Nil optional fields retain native defaults; pointers distinguish an explicit
// zero or false from an omitted setting. Configuration is copied during creation.
type TerminalConfig struct {
	Cols, Rows         uint16
	ScrollbackMaxBytes *uint   // Zero disables scrollback immediately.
	ScrollbackMaxLines *uint   // Pruned at native page boundaries, not an exact row cap.
	BackgroundColor    *uint32 // Default 0xRRGGBB.
	ForegroundColor    *uint32 // Default 0xRRGGBB.
	CursorColor        *uint32 // Default 0xRRGGBB.
	CursorBlink        *bool
	ModeDefaults       []ModeDefault
}

// ModeDefault sets both the current value and reset default of a mode.
type ModeDefault struct {
	Mode    Mode
	Enabled bool
}

// NewTerminalWithConfig creates and configures a terminal. The settings
// ghostty's own Options carries -- scrollback and the cursor defaults -- are
// passed to the constructor, so they hold from the first byte fed rather than
// from a setter afterwards. Colors and mode defaults cannot cross at
// construction and are applied after; a failure in one of those closes the
// newly created terminal, so no partially configured handle escapes.
func NewTerminalWithConfig(cfg TerminalConfig, extra ...TerminalOption) (_ *Terminal, err error) {
	if cfg.Cols == 0 {
		return nil, &RangeError{Operation: "NewTerminalWithConfig", Parameter: "Cols", Type: "nonzero cell count"}
	}
	if cfg.Rows == 0 {
		return nil, &RangeError{Operation: "NewTerminalWithConfig", Parameter: "Rows", Type: "nonzero cell count"}
	}
	for _, mode := range cfg.ModeDefaults {
		if !mode.Mode.IsKnown() {
			return nil, &RangeError{Operation: "NewTerminalWithConfig", Parameter: "ModeDefaults", Type: "known Mode"}
		}
	}

	// A nil field means "leave ghostty's default", which is what omitting the
	// option does, so the nil checks decide whether to append rather than what
	// to pass.
	opts := make([]TerminalOption, 0, 3+len(extra))
	if cfg.ScrollbackMaxBytes != nil {
		opts = append(opts, WithTerminalMaxScrollbackBytes(cfg.ScrollbackMaxBytes))
	}
	if cfg.ScrollbackMaxLines != nil {
		opts = append(opts, WithTerminalMaxScrollbackLines(cfg.ScrollbackMaxLines))
	}
	if cfg.CursorBlink != nil {
		opts = append(opts, WithTerminalDefaultCursorBlink(cfg.CursorBlink))
	}
	opts = append(opts, extra...)

	term, err := NewTerminal(cfg.Cols, cfg.Rows, opts...)
	if err != nil {
		return nil, err
	}
	defer func() {
		if err != nil {
			_ = term.Close()
		}
	}()
	if cfg.BackgroundColor != nil {
		if err = term.SetDefaultBackgroundColor(*cfg.BackgroundColor); err != nil {
			return nil, err
		}
	}
	if cfg.ForegroundColor != nil {
		if err = term.SetDefaultForegroundColor(*cfg.ForegroundColor); err != nil {
			return nil, err
		}
	}
	if cfg.CursorColor != nil {
		if err = term.SetDefaultCursorColor(*cfg.CursorColor); err != nil {
			return nil, err
		}
	}
	for _, mode := range cfg.ModeDefaults {
		if err = term.SetDefaultMode(mode.Mode, mode.Enabled); err != nil {
			return nil, err
		}
	}
	return term, nil
}

// StreamConfig configures the parser and its embedder identity. Zero limits
// disable continuation tracking and unknown-sequence capture respectively.
// A nil Version leaves Ghostty's identity unchanged. Callbacks run synchronously;
// only request-response calls may reenter the active terminal's bindings.
type StreamConfig struct {
	ContinuationMaxBytes uint
	UnknownMaxBytes      uint
	Version              *VersionReport
	EnquiryResponse      string
	ColorScheme          *ColorScheme
	ClipboardRead        ClipboardHandler
	ClipboardWrite       ClipboardHandler
}

// VersionReport is the application identity returned by XTVERSION.
type VersionReport struct{ Name, Version string }

// NewStreamWithConfig creates a child stream and applies all settings before
// returning it. On failure it closes the stream and releases the parent claim.
func (t *Terminal) NewStreamWithConfig(cfg StreamConfig) (_ *Stream, err error) {
	s, err := t.NewStream(cfg.ContinuationMaxBytes)
	if err != nil {
		return nil, err
	}
	defer func() {
		if err != nil {
			_ = s.Close()
		}
	}()
	if err = s.SetUnknownMaxBytes(cfg.UnknownMaxBytes); err != nil {
		return nil, err
	}
	if cfg.Version != nil {
		if err = s.SetVersionReport(cfg.Version.Name, cfg.Version.Version); err != nil {
			return nil, err
		}
	}
	if err = s.SetEnquiryResponse(cfg.EnquiryResponse); err != nil {
		return nil, err
	}
	if cfg.ColorScheme != nil {
		if *cfg.ColorScheme != ColorSchemeDark && *cfg.ColorScheme != ColorSchemeLight {
			return nil, &RangeError{Operation: "Terminal.NewStreamWithConfig", Parameter: "ColorScheme", Type: "ColorScheme"}
		}
		if err = s.ColorSchemeChanged(*cfg.ColorScheme); err != nil {
			return nil, err
		}
	}
	if cfg.ClipboardRead != nil {
		if err = s.OnClipboardReadRequest(cfg.ClipboardRead); err != nil {
			return nil, err
		}
	}
	if cfg.ClipboardWrite != nil {
		if err = s.OnClipboardWriteRequest(cfg.ClipboardWrite); err != nil {
			return nil, err
		}
	}
	return s, nil
}

// The option constructors below take values where the generated ones take
// pointers. `?usize` and `?bool` in ghostty's Options are three-state -- unset,
// or set to a number that may be zero -- so the generated `With*` spell that
// with a pointer. A caller who is calling the option at all has a value in
// hand, and `&` on a literal is not Go one can write, so these wrap it.

// WithScrollbackMaxBytes sets the maximum scrollback allocation in bytes.
// Zero disables scrollback immediately.
func WithScrollbackMaxBytes(bytes uint) TerminalOption {
	return WithTerminalMaxScrollbackBytes(&bytes)
}

// WithScrollbackMaxLines sets the maximum number of physical lines retained in
// scrollback. Pruned at native page boundaries, not an exact row cap.
func WithScrollbackMaxLines(lines uint) TerminalOption {
	return WithTerminalMaxScrollbackLines(&lines)
}

// WithDefaultCursorBlink sets the cursor blink state restored by DECSCUSR
// default and RIS.
func WithDefaultCursorBlink(blink bool) TerminalOption {
	return WithTerminalDefaultCursorBlink(&blink)
}

// MustNewTerminal creates a terminal with the given options, panicking on failure.
func MustNewTerminal(cols, rows uint16, opts ...TerminalOption) *Terminal {
	term, err := NewTerminal(cols, rows, opts...)
	if err != nil {
		panic(err)
	}
	return term
}

// StreamOption configures a Stream.
type StreamOption func(*StreamConfig)

// WithContinuationMaxBytes sets the maximum number of replay-safe VT continuation bytes retained.
func WithContinuationMaxBytes(bytes uint) StreamOption {
	return func(c *StreamConfig) { c.ContinuationMaxBytes = bytes }
}

// WithUnknownMaxBytes sets the maximum number of bytes retained for an unsupported escape sequence.
func WithUnknownMaxBytes(bytes uint) StreamOption {
	return func(c *StreamConfig) { c.UnknownMaxBytes = bytes }
}

// WithVersionReport configures the application identity returned by XTVERSION.
func WithVersionReport(name, version string) StreamOption {
	return func(c *StreamConfig) { c.Version = &VersionReport{Name: name, Version: version} }
}

// WithEnquiryResponse configures the response to ENQ (0x05).
func WithEnquiryResponse(response string) StreamOption {
	return func(c *StreamConfig) { c.EnquiryResponse = response }
}

// WithColorScheme sets the terminal color scheme (ColorSchemeDark or ColorSchemeLight).
func WithColorScheme(scheme ColorScheme) StreamOption {
	return func(c *StreamConfig) { c.ColorScheme = &scheme }
}

// WithClipboardReadHandler registers a handler for clipboard read requests.
func WithClipboardReadHandler(handler ClipboardHandler) StreamOption {
	return func(c *StreamConfig) { c.ClipboardRead = handler }
}

// WithClipboardWriteHandler registers a handler for clipboard write requests.
func WithClipboardWriteHandler(handler ClipboardHandler) StreamOption {
	return func(c *StreamConfig) { c.ClipboardWrite = handler }
}

// NewStreamWithOptions creates a child stream and applies functional options before returning it.
func (t *Terminal) NewStreamWithOptions(opts ...StreamOption) (*Stream, error) {
	var cfg StreamConfig
	for _, opt := range opts {
		opt(&cfg)
	}
	return t.NewStreamWithConfig(cfg)
}

// FormatString formats the active screen area according to opts and returns the result as a string.
func (t *Terminal) FormatString(opts FormatOptions) (string, error) {
	var buf strings.Builder
	if err := t.Format(opts, &buf); err != nil {
		return "", err
	}
	return buf.String(), nil
}

// PlainText returns the active screen area formatted as plain text.
// If trim is true, trailing whitespace on non-empty lines is omitted.
func (t *Terminal) PlainText(trim bool) (string, error) {
	return t.FormatString(FormatOptions{
		Format:                  FormatterFormatPlain,
		KeepTrailingWhitespace: !trim,
	})
}

// FormatString formats the screen contents (including scrollback) according to opts and returns a string.
func (s *Screen) FormatString(opts FormatOptions) (string, error) {
	var buf strings.Builder
	if err := s.Format(opts, &buf); err != nil {
		return "", err
	}
	return buf.String(), nil
}

// PlainText returns the screen contents (including scrollback) formatted as plain text.
// If trim is true, trailing whitespace on non-empty lines is omitted.
func (s *Screen) PlainText(trim bool) (string, error) {
	return s.FormatString(FormatOptions{
		Format:                  FormatterFormatPlain,
		KeepTrailingWhitespace: !trim,
	})
}

// SessionConfig configures both the terminal a session owns and its primary
// stream. TerminalOptions are the generated constructor options, applied after
// the fields TerminalConfig maps to the same options, so they win on conflict.
type SessionConfig struct {
	Terminal        TerminalConfig
	TerminalOptions []TerminalOption
	Stream          StreamConfig
}

// SessionOption configures a SessionConfig.
type SessionOption func(*SessionConfig)

// WithTerminalConfig sets the underlying TerminalConfig.
func WithTerminalConfig(cfg TerminalConfig) SessionOption {
	return func(c *SessionConfig) { c.Terminal = cfg }
}

// WithStreamConfig sets the underlying StreamConfig.
func WithStreamConfig(cfg StreamConfig) SessionOption {
	return func(c *SessionConfig) { c.Stream = cfg }
}

// WithTerminalOptions adds constructor options for the session's terminal.
// A TerminalOption is the generated kind, which the constructor consumes; it
// cannot be applied to a TerminalConfig, so it is collected rather than run here.
func WithTerminalOptions(opts ...TerminalOption) SessionOption {
	return func(c *SessionConfig) {
		c.TerminalOptions = append(c.TerminalOptions, opts...)
	}
}

// WithStreamOptions applies functional options to the session's stream.
func WithStreamOptions(opts ...StreamOption) SessionOption {
	return func(c *SessionConfig) {
		for _, opt := range opts {
			opt(&c.Stream)
		}
	}
}

// NewConfiguredSession creates a terminal and a primary stream wired to it, and
// adopts both into a session. NewSession adopts an already-built terminal; this
// builds the pair from a configuration. If either step fails, nothing escapes:
// whatever was allocated is closed before the error is returned.
func NewConfiguredSession(cols, rows uint16, opts ...SessionOption) (*Session, error) {
	cfg := SessionConfig{Terminal: TerminalConfig{Cols: cols, Rows: rows}}
	for _, opt := range opts {
		opt(&cfg)
	}
	term, err := NewTerminalWithConfig(cfg.Terminal, cfg.TerminalOptions...)
	if err != nil {
		return nil, err
	}
	stream, err := term.NewStreamWithConfig(cfg.Stream)
	if err != nil {
		_ = term.Close()
		return nil, err
	}
	return NewSession(term).AddStream(stream), nil
}

// MustNewConfiguredSession creates a configured session, panicking on failure.
func MustNewConfiguredSession(cols, rows uint16, opts ...SessionOption) *Session {
	sess, err := NewConfiguredSession(cols, rows, opts...)
	if err != nil {
		panic(err)
	}
	return sess
}

// Stream returns the session's primary stream: the first one adopted, which is
// the one NewConfiguredSession wired to the terminal. It is nil for a session
// that has adopted no stream, and after Close.
func (s *Session) Stream() *Stream {
	streams := s.Streams()
	if len(streams) == 0 {
		return nil
	}
	return streams[0]
}

// Write feeds bytes into the session's primary stream, satisfying io.Writer.
func (s *Session) Write(p []byte) (int, error) {
	stream := s.Stream()
	if stream == nil {
		return 0, &HandleError{Operation: "Session.Write"}
	}
	return stream.Write(p)
}

// WriteString feeds str into the session's primary stream, satisfying io.StringWriter.
func (s *Session) WriteString(str string) (int, error) {
	stream := s.Stream()
	if stream == nil {
		return 0, &HandleError{Operation: "Session.WriteString"}
	}
	return stream.WriteString(str)
}

// FormatString formats the active screen area and returns the result as a string.
func (s *Session) FormatString(opts FormatOptions) (string, error) {
	return s.Terminal().FormatString(opts)
}

// PlainText returns the active screen area formatted as plain text.
func (s *Session) PlainText(trim bool) (string, error) {
	return s.Terminal().PlainText(trim)
}

var (
	_ io.Writer       = (*Session)(nil)
	_ io.StringWriter = (*Session)(nil)
	_ io.Closer       = (*Session)(nil)
	_ io.StringWriter = (*Stream)(nil)
)
